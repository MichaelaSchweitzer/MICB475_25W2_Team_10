## !/bin/bash

# log onto server with team 10 credentials (swap * for IP, enter password when prompted) 
ssh root@*

# march 21st, 2026, madi 
# picrust analysis (server part, following module 19 tutorial) 
# move to folder for fish microbiome data processing
cd /data/fish_microbiome


# filter out features with fewer than 5 counts to speed up analysis 
qiime feature-table filter-features \
  --i-table table.qza \
  --p-min-frequency 5 \
  --o-filtered-table feature-frequency-filtered-table.qza

# export data 
mkdir picrust
qiime tools export \
   --input-path feature-frequency-filtered-table.qza \
   --output-path picrust
qiime tools export \
   --input-path representative-sequences.qza \
   --output-path picrust

# switch to picrust environment 
conda deactivate
conda activate picrust2

# run picrust
picrust2_pipeline.py \
   -s picrust/dna-sequences.fasta \
   -i picrust/feature-table.biom \
   -o picrust_out

# export file (in terminal not connected to server)
# navigate to github clone directory 
# swap * for IP, enter password when prompted
scp -r root@*:/data/fish_microbiome/picrust_out .  
# proceed to R analysis (aim_4_functional.R)
