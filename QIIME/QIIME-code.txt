####Denoiseing and declustering using ASVs with DADA2 (for project 2 proposal)
# Determine ASVs with DADA2
#done in data/fish_microbiome 
#chose this truncation length of 218 as after 218 almost avery bp is below 25 phred score in QIIME2 online viewer of exported demux.qzv. Consulted Michaela, Madi, and Mirren. 
  qiime dada2 denoise-single \
    --i-demultiplexed-seqs single-end-demux.qza \
    --p-trim-left 0 \
    --p-trunc-len 218 \
    --o-representative-sequences representative-sequences.qza \
    --o-table table.qza \
    --o-denoising-stats denoising-stats.qza
  
      #Output: 
      #Saved FeatureTable[Frequency] to: table.qza
      #Saved FeatureData[Sequence] to: representative-sequences.qza
      #Saved SampleData[DADA2Stats] to: denoising-stats.qza

# Visualize ASVs stats (for addition of information on the denoised dataset to project 2 proposal) # Ran in fish_microbiome
qiime feature-table tabulate-seqs \
  --i-data representative-sequences.qza \
  --o-visualization rep-seqs.qzv

  # Output: Saved Visualization to: rep-seqs.qzv

 qiime feature-table summarize \
  --i-table table.qza \
  --o-visualization table.qzv \
  --m-sample-metadata-file /datasets/project_2/fish/fish_metadata.txt

  # Output: Saved Visualization to: table.qzv

######Clustering and Taxonometric analysis 

# Make a new file to store converted demultiplexed fish dataset analysis in for export

# Taxonomic analysis - used the database pre-loaded onto the server for V4 region. Note that the Minich et al BioArchive paper did their analysis on V4 as well so this should work! 
qiime feature-classifier classify-sklearn \
  --i-classifier /datasets/classifiers/silva-138-99-515-806-nb-classifier.qza \
  --i-reads representative-sequences.qza \
  --o-classification taxonomy.qza

# Output: Saved FeatureData[Taxonomy] to: taxonomy.qza

qiime metadata tabulate \
  --m-input-file taxonomy.qza \
  --o-visualization taxonomy.qzv
  
# Output: Saved Visualization to: taxonomy.qzv

# Taxonomy barplots

  --i-table table.qza \
  --i-taxonomy taxonomy.qza \
  --m-metadata-file /datasets/project_2/fish/fish_metadata.txt \
  --o-visualization taxa-bar-plots.qzv

### alpha-rarefaction 
 
cd /data/fish_microbiome

# generate tree for phylogenetic diversity analyses
qiime phylogeny align-to-tree-mafft-fasttree \
  --i-sequences representative-sequences.qza \
  --o-alignment aligned-rep-seqs.qza \
  --o-masked-alignment masked-aligned-rep-seqs.qza \
  --o-tree unrooted-tree.qza \
  --o-rooted-tree rooted-tree.qza 

# alpha-rarefaction
qiime diversity alpha-rarefaction \
  --i-table table.qza \
  --i-phylogeny rooted-tree.qza \
  --p-max-depth 24641 \
  --m-metadata-file /datasets/project_2/fish/fish_metadata.txt \
  --o-visualization alpha-rarefaction.qzv

# export visualization file (in terminal not connected to server, enter password when prompted)
scp root@*:/data/fish_microbiome/alpha-rarefaction.qzv .
