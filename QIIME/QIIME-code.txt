# import fish data

  --type 'SampleData[SequencesWithQuality]' \
  --input-path /datasets/project_2/fish/fish_manifest.tsv \
  --output-path single-end-demux.qza \
  --input-format SingleEndFastqManifestPhred33V2

# generate visualization file 
qiime demux summarize \
  --i-data single-end-demux.qza \
  --o-visualization demux.qzv

# export file (in terminal not connected to server, enter password when prompted)

####Denoiseing and declustering using ASVs with DADA2 (for project 2 proposal)
# Determine ASVs with DADA2
#done in data/fish_microbiome 
#chose this truncation length of 218 as after 218 almost avery bp is below 25 phred score in QIIME2 online viewer of exported demux.qzv. Consulted Michaela, Madi, and Mirren. 
  qiime dada2 denoise-single \
    --i-demultiplexed-seqs single-end-demux.qza \
    --p-trim-left 0 \
    --p-trunc-len 218 \
    --o-representative-sequences representative-sequences.qza \
    --o-table table.qza \
    --o-denoising-stats denoising-stats.qza
  
      #Output: 
      #Saved FeatureTable[Frequency] to: table.qza
      #Saved FeatureData[Sequence] to: representative-sequences.qza
      #Saved SampleData[DADA2Stats] to: denoising-stats.qza

# Visualize ASVs stats (for addition of information on the denoised dataset to project 2 proposal) # Ran in fish_microbiome
qiime feature-table tabulate-seqs \
  --i-data representative-sequences.qza \
  --o-visualization rep-seqs.qzv

  # Output: Saved Visualization to: rep-seqs.qzv

 qiime feature-table summarize \
  --i-table table.qza \
  --o-visualization table.qzv \
  --m-sample-metadata-file /datasets/project_2/fish/fish_metadata.txt

  # Output: Saved Visualization to: table.qzv

##Export visualizations (run each individually in local computer)
